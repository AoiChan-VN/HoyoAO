├── assets/
│   ├── skybox/                     # Các lớp ảnh nền (.webp cho nền rộng, .png cho vật thể lơ lửng)
│   └── posts/                      # Thư mục chứa các bài viết Markdown (.md) nội bộ
├── domain/                         # TẦNG LÕI TOÁN HỌC & ĐỒ HỌA THUẦN (Enterprise Core Logic)
│   ├── Matrix4.js                  # Toán học ma trận 4x4, Perspective, View Matrix 3D VR
│   └── Vector3.js                  # Tính toán Vector, hướng nhìn trong không gian 3D
├── infrastructure/                 # TẦNG GIAO TIẾP PHẦN CỨNG & THIẾT BỊ (Hardware & Secure Adapters)
│   ├── SensorCamera.js             # Chuột + Gyroscope siêu mượt với Complementary Filter khử trễ/nhiễu
│   ├── WebGLRenderer.js            # WebGL2 thuần xử lý Phong Lighting (Ánh nắng đổi hướng theo góc nhìn thực)
│   └── MarkdownSanitizer.js        # Bộ Parser .md sang HTML bằng Regex thuần tích hợp Sanitize bảo mật quốc tế
├── application/                    # TẦNG ĐIỀU PHỐI VÀ QUẢN LÝ (Orchestration & State)
│   ├── AppController.js            # Quản lý Main Loop, trạng thái đóng/mở Panel, đồng bộ Camera & WebGL
│   └── PerformanceRegistry.js      # Khóa FPS (60 - 120 - Uncapped đồng bộ phần cứng máy)
└── presentation/                   # TẦNG UI / HUD / INTERACTION (View & Physics Presentation)
    ├── core/
    │   └── PhysicsDragDrop.js      # Engine vật lý kéo thả tự do, tạo góc nghiêng quán tính khi di chuyển UI
    ├── components/
    │   ├── CentralMenu.js          # Menu chính gốc tọa độ (HUD Center) - Không Drag. Chỉ chứa các nút kích hoạt
    │   ├── SettingsPanel.js        # Panel cài đặt hiệu năng (FPS, Đồ họa) - Có Drag-Drop vật lý
    │   ├── PostListPanel.js        # Panel hiển thị danh sách bài viết .md - Có Drag-Drop vật lý
    │   └── ReaderPanel.js          # Panel đọc nội dung chi tiết bài viết - Có Drag-Drop vật lý
    ├── css/
    │   └── layout3d.css            # Không gian preserve-3d, chiều sâu tầng lớp Infinity Skybox
    └── index.html                  # Điểm neo ứng dụng (DOM entry)
